cmake_minimum_required(VERSION 3.16)

project(SGP4 VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB_RECURSE ALL_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
)

list(FILTER ALL_SOURCES EXCLUDE REGEX "/build/")
list(FILTER ALL_SOURCES EXCLUDE REGEX ".*cmake-build-debug.*")
list(FILTER ALL_SOURCES EXCLUDE REGEX ".*CMakeFiles.*")

add_executable(SGP4 ${ALL_SOURCES})
